Code.gs — paste this into Apps Script 
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var d = JSON.parse(e.postData.contents);
  if (sheet.getLastRow() === 0) {
    sheet.appendRow([
      'Timestamp', 'Name', 'Email', 'Position', 'Language',
      'PS Score', 'PS %', 'TM Score', 'TM %', 'Comm Score', 'Comm %',
      'OCEAN Answered', 'E%', 'A%', 'C%', 'N%', 'O%'
    ]);
  }
  sheet.appendRow([
    d.timestamp, d.name, d.email, d.position, d.language,
    d.ps_score, d.ps_pct, d.tm_score, d.tm_pct, d.comm_score, d.comm_pct,
    d.ocean_answered, d.ocean_E, d.ocean_A, d.ocean_C, d.ocean_N, d.ocean_O
  ]);
  return ContentService
    .createTextOutput(JSON.stringify({ status: 'ok' }))
    .setMimeType(ContentService.MimeType.JSON);
}
💡 After pasting, save with Ctrl + S (Windows) or Cmd + S (Mac).
